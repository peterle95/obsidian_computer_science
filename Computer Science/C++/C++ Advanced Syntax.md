// Get all flashcards from the current note
const currentPage = dv.current();
const content = await app.vault.read(app.vault.getAbstractFileByPath(currentPage.file.path));

// Split content into lines
const lines = content.split('\n');
let flashcardLines = [];
let inCodeBlock = false;

// Collect all potential flashcard lines - simplified approach
for (let i = 0; i < lines.length; i++) {
    const line = lines[i];
    
    // Track code blocks
    if (line.trim().startsWith('```')) {
        inCodeBlock = !inCodeBlock;
        continue;
    }
    
    // Skip everything inside code blocks
    if (inCodeBlock) continue;
    
    // Check if line contains flashcard separator
    if (line.includes(';;')) {
        flashcardLines.push(line);
    }
}

const filteredLines = flashcardLines.filter(line => {
    // Only filter out lines that are clearly part of the JavaScript code
    // Be more specific with patterns to avoid false positives
    return !(
        line.trim().startsWith('const ') ||
        line.trim().startsWith('let ') ||
        line.trim().startsWith('function ') ||
        line.trim().startsWith('return ') ||
        line.trim().startsWith('if (') ||
        line.trim().startsWith('for (') ||
        line.trim().startsWith('while (') ||
        line.includes('dataviewjs') ||
        line.includes('content.split') ||
        line.includes('flashcardLines') ||
        line.includes('this.container') ||
        line.includes('addEventListener') ||
        line.includes('console.log') ||
        /\.\w+\(/.test(line) && (line.includes('.map(') || line.includes('.filter(') || line.includes('.forEach(') || line.includes('.find('))
    );
});

// Process the flashcards
const flashcards = [];

    const line = filteredLines[i];
    try {
        const separatorIndex = line.indexOf(';;');
        if (separatorIndex === -1) continue;
        
        const front = line.substring(0, separatorIndex).trim();
        const back = line.substring(separatorIndex + 2).trim();
        
        // Very minimal validation - just check they exist
        if (front && back) {
            flashcards.push({ 
                front: front, 
                back: back,
                index: i // Keep track of original order
            });
        }
    } catch (error) {
        console.log('Error processing flashcard line:', line, error);

# **Core Explanation:**

## Constructor Initialization List

### Basic Syntax
```cpp
ClassName(parameters) : member1(value1), member2(value2) { }
//        ^             ^
//        Parameters    Initialization list
```

### Examples
```cpp
// Simple initialization
Person(std::string n, int a) : name(n), age(a) {}
// Means: Initialize 'name' with 'n' and 'age' with 'a'

// Pointer initialization
ResourceHandler() : ptr(new Resource()) {}
// Means: Initialize 'ptr' with a new Resource object

// Multiple initializations
Complex(double r, double i) : real(r), imag(i), count(0) {}
// Means: Initialize multiple members at once
```

## Destructors

### Basic Syntax
```cpp
~ClassName() { /* cleanup code */ }
//^
//Tilde indicates destructor
```

### Examples
```cpp
// Simple destructor
~Person() {}

// Destructor with cleanup
~ResourceHandler() { delete ptr; }
// Means: When object is destroyed, delete the pointed-to Resource

// Virtual destructor
virtual ~BaseClass() {}
// Means: Allow proper cleanup in inheritance
```

## Static Members

### Basic Syntax
```cpp
static returnType memberName;  // Declaration
static returnType functionName();  // Function declaration
```

### Examples
```cpp
// Static variable
static int count;  // Shared among all instances

// Static function
static void reset();  // Can be called without an instance

// Static pointer
static Singleton* instance;  // Shared pointer among all instances
```

## Member Function Declaration Patterns

### Basic Syntax
```cpp
returnType functionName(parameters) specifiers;
```

### Examples
```cpp
// Basic member function
void setName(std::string name);

// Const member function
int getAge() const;  // Won't modify object

// Virtual function
virtual void draw();  // Can be overridden

// Pure virtual function
virtual void process() = 0;  // Must be overridden
```

## Pointer and Reference Syntax

### Basic Syntax
```cpp
Type* pointerName;    // Pointer declaration
Type& referenceName;  // Reference declaration
```

### Examples
```cpp
// Pointer to object
Person* ptr = new Person();

// Reference parameter
void update(Person& person);

// Const reference
void display(const Person& person);
```

## Common Complex Patterns

### 1. Constructor with Dynamic Allocation
```cpp
class Handler {
private:
    Resource* ptr;
public:
    // Constructor allocates memory
    Handler() : ptr(new Resource()) {}
    //          ^Initialization list allocates new object
    
    // Destructor frees memory
    ~Handler() { delete ptr; }
    //          ^Cleanup allocated memory
};
```

### 2. Singleton Pattern
```cpp
class Singleton {
private:
    static Singleton* instance;    // Static pointer member
    Singleton() {}                 // Private constructor
    
public:
    static Singleton* getInstance() {  // Static access method
        if (!instance) {
            instance = new Singleton();
        }
        return instance;
    }
};
```

### 3. Member Function Modifiers
```cpp
class Example {
public:
    // const: Cannot modify object
    int getValue() const { return value; }
    
    // virtual: Can be overridden
    virtual void process() { }
    
    // override: Is overriding base class
    void draw() override { }
    
    // final: Cannot be further overridden
    virtual void render() final { }
};
```

## Special Syntax Markers

### 1. Scope Resolution Operator (::)
```cpp
class MyClass {
public:
    static int count;
};

// Define static member
int MyClass::count = 0;
//   ^Scope resolution operator

// Define member function outside class
void MyClass::function() { }
```

### 2. Member Access Operators
```cpp
MyClass object;
object.member;    // Dot operator for objects

MyClass* ptr = new MyClass();
ptr->member;      // Arrow operator for pointers
```

### 3. Initialization Patterns
```cpp
// Uniform initialization
MyClass obj{10, "Hello"};

// Direct initialization
MyClass obj(10, "Hello");

// Copy initialization
MyClass obj = other;
```

Remember:
- The colon (:) in constructors starts the initialization list
- Tilde (~) indicates a destructor
- Static members belong to the class, not instances
- Arrow (->) is for pointer access
- Dot (.) is for direct member access
- Double colon (::) is for scope resolution
