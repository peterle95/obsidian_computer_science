// Collect all potential flashcard lines - simplified approach

    const line = lines[i];

    if (line.trim().startsWith('```')) {
        inCodeBlock = !inCodeBlock;
        continue;
    }
    
    // Skip everything inside code blocks

    // Check if line contains flashcard separator
    if (line.includes(';;')) {
        flashcardLines.push(line);
    }
}

    // Only filter out lines that are clearly part of the JavaScript code
    // Be more specific with patterns to avoid false positives
    return !(
        line.trim().startsWith('const ') ||
        line.trim().startsWith('let ') ||
        line.trim().startsWith('function ') ||
        line.trim().startsWith('return ') ||

# **Exercise**:

A **pangram** is a sentence where every letter of the English alphabet appears at least once.

Given a string `sentence` containing only lowercase English letters, return `true` _if_ `sentence` _is a **pangram**, or_ `false` _otherwise._

**Example 1:**

**Input:** sentence = "thequickbrownfoxjumpsoverthelazydog"
**Output:** true
**Explanation:** sentence contains at least one of every letter of the English alphabet.

**Example 2:**

**Input:** sentence = "leetcode"
**Output:** false

**Constraints:**

- `1 <= sentence.length <= 1000`
- `sentence` consists of lowercase English letters.

# **Starting code:**
---

```js
/**
 * @param {string} sentence
 * @return {boolean}
 */
var checkIfPangram = function(sentence) {
    
};
```


# **Solution:**
---
```js
/**
 * @param {string} sentence
 * @return {boolean}
 */
var checkIfPangram = function(sentence) {
    const alphabet = new Set('abcdefghijklmnopqrstuvwxyz');
    
    for (let char of sentence) {
        alphabet.delete(char);
        if (alphabet.size === 0) 
	        return true;
    }
    
    return alphabet.size === 0;
};
```
# **Explanation:**
---

### **Approach: Set-Based Character Tracking**

The solution uses a Set data structure to track which letters of the alphabet we still need to find in the sentence.

### **Step 1: Initialize the Alphabet Set**

```js
const alphabet = new Set('abcdefghijklmnopqrstuvwxyz');
```

- Create a Set containing all 26 lowercase letters of the English alphabet 
- A Set is perfect here because it allows fast deletion and size checking
- Initial size: 26 letters
- [[JavaScript Map and Set (ES6)]]

### **Step 2: Iterate Through the Sentence**

```js
for (let char of sentence) {
```

- Loop through each character in the input sentence
- We'll check each character to see if it's a letter we're still looking for
- [[JavaScript for...of loop revisited & Iterators]]
- [[JavaScript Loops (for, while, do...while, for...in, for...of)]]

### **Step 3: Remove Found Letters**

```js
alphabet.delete(char);
```

- When we encounter a character, remove it from the `alphabet` Set
- If the character is already removed (duplicate letter), `delete()` does nothing
- This gradually reduces the Set size as we find each unique letter

### **Step 4: Early Exit Optimization**

```js
if (alphabet.size === 0) 
    return true;
```

- After each deletion, check if the Set is empty 
- If `size === 0`, we've found all 26 letters—it's a pangram!
- This optimization allows us to return early without processing the entire sentence

### **Step 5: Final Check**

```js
return alphabet.size === 0;
```

- If we finish the loop without early exit, check the Set size one final time
- Returns `true` if all letters were found (size = 0)
- Returns `false` if any letters remain in the Set

### **Example Walkthrough**

**Input:** `"thequickbrownfoxjumpsoverthelazydog"`

1. Start with Set of 26 letters
2. Process 't' → delete 't' → 25 letters remain
3. Process 'h' → delete 'h' → 24 letters remain
4. Continue until all 26 unique letters are found
5. Set becomes empty → return `true`

**Input:** `"leetcode"`

1. Start with Set of 26 letters
2. Process and delete: 'l', 'e', 't', 'c', 'o', 'd'
3. After entire string: 20 letters still remain in Set
4. Set is not empty → return `false`

### **Time & Space Complexity**

- **Time:** $O(n)$ where $n$ is the length of the sentence (worst case: scan entire string)
- **Space:** $O(1)$ constant space (Set always contains at most 26 letters)
